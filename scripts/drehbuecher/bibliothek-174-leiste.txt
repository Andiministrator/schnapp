# scripts/probelibrary legt 14 synthetische Aufnahmen über heute, gestern, die letzten Wochen und das Vorjahr an, dazu

#   bild=$(go run ./scripts/probelibrary -heim tmp/probe/home)

#   scripts/probe.sh --drehbuch scripts/drehbuecher/bibliothek-174-leiste.txt --ende 60 --bild "$bild"
